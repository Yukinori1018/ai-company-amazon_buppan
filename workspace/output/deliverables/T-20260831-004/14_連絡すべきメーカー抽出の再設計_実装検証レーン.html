def independent_sellers(brand_norm, sellers):
    """卸証拠に数えてよいセラーだけを返す。"""
    out = []
    for x in sellers:
        if SELLER_ID.match(x.strip()):  continue   # 名前が未解決 → 別枠で扱う
        if DIRECT_SHOP.search(x):       continue   # 直営/公式ストア/正規販売店
        n = nrm(x)
        if brand_norm and len(brand_norm) >= 3 and (brand_norm in n or n in brand_norm):
            continue                               # ブランド名を冠する系列店
        out.append(x)
    return out
